\section{Complete Mathematical Proofs}
\label{app:proofs}

In this appendix, we provide complete derivations and proofs for Proposition~\ref{prop:subspace}, Theorem~\ref{thm:kkt}, Theorem~\ref{thm:invariance}, and Theorem~\ref{thm:shrinkage}.

\subsection{Proof of Proposition~\ref{prop:subspace} (Adjoint Decomposition)}
\label{app:proof_prop1}

\begin{proof}
Let $\{x_i^\star\}_{i=0}^T$ satisfy $x_{i+1}^\star = a_i x_i^\star + b_i v_i^\star$ with $v_i^\star = f_\omega(x_i^\star, \tau_i, c)$, and let $\{x_i\}_{i=0}^T$ be generated under schedule $\cS = \{s_0, \dots, s_{B-1}\}$ ($s_0 = 0, s_{B-1} = T-1$) and predictor $\cR$, satisfying $x_{i+1} = a_i x_i + b_i \tilde{v}_i$ with $x_0 = x_0^\star$. Set $\delta x_i \coloneqq x_i - x_i^\star$ and $J_i \coloneqq \nabla_x f_\omega(x_i^\star, \tau_i, c)$.

\textbf{Step 1 (Closed-loop horizon unrolling).}
Fix an interval $\mathcal{I}_m = \{s_m, \dots, s_{m+1} - 1\}$ of gap $g_m = s_{m+1} - s_m \ge 1$. At exact step $i = s_m \in \cS$, first-order Taylor expansion around $x_{s_m}^\star$ gives
\begin{equation}
\tilde{v}_{s_m} - v_{s_m}^\star = J_{s_m} \delta x_{s_m} + \mathcal{O}(\norm{\delta x_{s_m}}_2^2),
\end{equation}
so $\delta x_{s_m + 1} = (a_{s_m} I + b_{s_m} J_{s_m})\delta x_{s_m} + \mathcal{O}(\norm{\delta x_{s_m}}_2^2) = A_{s_m}^{\mathrm{cl}} \delta x_{s_m} + \mathcal{O}(\norm{\delta x_{s_m}}_2^2)$. Across skipped steps $j = s_m + k$ ($1 \le k \le g_m - 1$), $\cR(\cB_{s_m}, j)$ depends only on cached anchors up to $s_m$, so unrolling $x_{j+1} - x_{j+1}^\star = a_j \delta x_j + b_j(\cR(\cB_{s_m}, j) - v_j^\star)$ across $k = 1, \dots, g_m - 1$ yields
\begin{equation}
\begin{aligned}
\delta x_{s_{m+1}} &= \Bigl(\prod_{\ell = s_m + 1}^{s_{m+1}-1} a_\ell\Bigr) A_{s_m}^{\mathrm{cl}} \delta x_{s_m} \\
&\quad + \sum_{k=1}^{g_m - 1} \omega_{m,k} \bigl(\cR(\cB_{s_m}, s_m + k) - v_{s_m + k}^\star\bigr),
\end{aligned}
\end{equation}
up to $\mathcal{O}(\norm{\delta x}_2^2)$, with $\omega_{m,k} = (\prod_{\ell=s_m+k+1}^{s_{m+1}-1} a_\ell) b_{s_m+k}$. Inducting over $m = 0, \dots, B - 2$ with $\delta x_0 = 0$ and closed-loop transition $\Psi_{s_{m+1}\to T} = \prod_{\ell=s_{m+1}}^{T-1} A_\ell^{\mathrm{cl}}$ yields Eq.~\eqref{eq:adjoint_sum}, and the chain rule on $\Phi \circ \cD$ at $x_T^\star$ gives $\Phi(\cD(x_T)) - \Phi(\cD(x_T^\star)) = J_{\Phi \circ \cD}(x_T^\star)\,\delta x_T + \mathcal{O}(\norm{\delta x_T}_2^2)$.

\textbf{Step 2 (Subspace orthogonality bound).}
Let $\mathcal{U}_m \coloneqq \mathrm{span}\{v : (\cdot, v) \in \cB_{s_m}\} \subseteq \R^d$. Decomposing $v_j^\star = \Pi_{\cB_{s_m}} v_j^\star + \Pi_{\cB_{s_m}}^\perp v_j^\star$ orthogonally in $\R^d$, for any $\cR_{\mathrm{lin}}(\cB_{s_m}, j) \in \mathcal{U}_m$ the Pythagorean theorem gives
\begin{equation}
\begin{aligned}
\norm{\cR_{\mathrm{lin}}(\cB_{s_m}, j) - v_j^\star}_2^2 &= \norm{\cR_{\mathrm{lin}}(\cB_{s_m}, j) - \Pi_{\cB_{s_m}} v_j^\star}_2^2 \\
&\quad + \norm{\Pi_{\cB_{s_m}}^\perp v_j^\star}_2^2 \ge \norm{\Pi_{\cB_{s_m}}^\perp v_j^\star}_2^2,
\end{aligned}
\end{equation}
with equality iff $\cR_{\mathrm{lin}}(\cB_{s_m}, j) = \Pi_{\cB_{s_m}} v_j^\star$, proving Eq.~\eqref{eq:subspace_floor}.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:kkt} (KKT Head \& Geometric Tail)}
\label{app:proof_thm1}

\begin{proof}
Fix $\beta > 0$, $B \in (0, T)$, and continuous strictly decreasing $\tilde{w} : [0, T] \to (0, \infty)$. For $f(\rho) \coloneqq \frac{1}{1+\beta}(\rho^{-\beta} - \rho)$ on $(0, 1]$:
\begin{equation}
f'(\rho) = -\frac{\beta \rho^{-(1+\beta)} + 1}{1 + \beta} < 0, \qquad f''(\rho) = \frac{\beta}{\rho^{2+\beta}} > 0.
\end{equation}
Since $\tilde{w}(t) > 0$, $\tilde{w}(t) f(\rho(t))$ is strictly convex, diverges as $\rho \downarrow 0$ (enforcing $\rho^\star(t) > 0$), and vanishes at $\rho = 1$. With dual multiplier $\lambda \in \R$ for $\int_0^T \rho(t)\,dt = B$ and $\mu(t) \ge 0$ for $\rho(t) - 1 \le 0$, necessary and sufficient KKT conditions are:
\begin{align}
-\frac{\tilde{w}(t)}{1+\beta}\bigl(\beta \rho^\star(t)^{-(1+\beta)} + 1\bigr) + \lambda^\star + \mu^\star(t) &= 0, \label{eq:kkt_stat}\\
0 < \rho^\star(t) \le 1,\;\; \mu^\star(t) \ge 0,\;\; \mu^\star(t)\bigl(\rho^\star(t) - 1\bigr) &= 0. \label{eq:kkt_cs}
\end{align}

\textbf{Part 1 (Head saturation and budget monotonicity).}
Where $\rho^\star(t) < 1$, complementary slackness gives $\mu^\star(t) = 0$, so
\begin{equation}
\beta \tilde{w}(t) \rho^\star(t)^{-(1+\beta)} = (1+\beta)\lambda^\star - \tilde{w}(t).
\end{equation}
Having $\rho^\star(t) \in (0, 1)$ requires $(1+\beta)\lambda^\star - \tilde{w}(t) > \beta\tilde{w}(t) \iff \tilde{w}(t) < \lambda^\star$, yielding $\rho^\star(t) = \bigl(\frac{\beta\tilde{w}(t)}{(1+\beta)\lambda^\star - \tilde{w}(t)}\bigr)^{\frac{1}{1+\beta}}$. Where $\tilde{w}(t) \ge \lambda^\star$, setting $\rho^\star(t) = 1$ and $\mu^\star(t) = \tilde{w}(t) - \lambda^\star \ge 0$ satisfies Eqs.~\eqref{eq:kkt_stat}--\eqref{eq:kkt_cs}. Since $\tilde{w}$ is strictly decreasing, $\{t : \tilde{w}(t) \ge \lambda^\star\} = [0, H^\star(B)]$. Moreover, $G(\lambda) \coloneqq \int_0^T \rho^\star(t; \lambda)\,dt$ is continuous and strictly decreasing for $\lambda \in (\tilde{w}(T), \infty)$ with $G(\tilde{w}(0)) \le T$ and $\lim_{\lambda \to \infty} G(\lambda) = 0$ (and $G(\lambda) = T$ for $\lambda \le \tilde{w}(T)$). Hence $\lambda^\star(B)$ is unique and strictly decreasing in $B \in (0, T)$, so $H^\star(B) = \min(T, \max(0, \tilde{w}^{-1}(\lambda^\star(B))))$ is non-decreasing in $B$. As $\beta \downarrow 0$, $f(\rho) \to 1 - \rho$ becomes affine and degenerates to $\rho^\star(t) = \mathbb{I}[t \le B]$.

\textbf{Part 2 (Growing tail).}
For $t > H^\star(B)$, $g^\star(t) = 1/\rho^\star(t) = \bigl(\frac{1+\beta}{\beta}\frac{\lambda^\star}{\tilde{w}(t)} - \frac{1}{\beta}\bigr)^{\frac{1}{1+\beta}}$ increases strictly with $t$. When $\tilde{w}(t) = w_0 e^{-\eta t} \ll (1+\beta)\lambda^\star$, $g^\star(t) \simeq C e^{\frac{\eta}{1+\beta} t}$, giving discrete ratio $g_{k+1}/g_k \approx e^{\eta/(1+\beta)} = r > 1$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:invariance} (Scale-Invariance Collapse)}
\label{app:proof_thm2}

\begin{proof}
We induct on the exact call index $k = 0, \dots, B - 2$. At $k=0$, $s_0 = 0 = s_0^\star$. Assume after $k$ exact calls the online controller matches the offline optimum prefix $(s_0^\star, \dots, s_k^\star)$. At step $s_k^\star$ with $B - k$ calls remaining, let $\mathfrak{S}_{B-k}(s_k^\star)$ denote strictly increasing sequences $(u_0, \dots, u_{B-1-k})$ with $u_0 = s_k^\star$ and $u_{B-1-k} = T-1$. Since $\hat{\sigma}_k(x_{0:s_k^\star}) > 0$ is constant wrt $(u_1, \dots, u_{B-2-k})$,
\begin{equation}
\begin{aligned}
&\arg\min_{u \in \mathfrak{S}_{B-k}(s_k^\star)} \sum_{m=0}^{B-2-k} \hat{\sigma}_k\, C_\beta(u_m, u_{m+1}) \\
&\qquad = \arg\min_{u \in \mathfrak{S}_{B-k}(s_k^\star)} \sum_{m=0}^{B-2-k} C_\beta(u_m, u_{m+1}).
\end{aligned}
\end{equation}
By Bellman's Principle of Optimality on $\cS^\star$, $(s_k^\star, s_{k+1}^\star, \dots, s_{B-1}^\star)$ solves the unscaled suffix problem, so the controller chooses $u_1 = s_{k+1}^\star$. By induction, $\cS_{\mathrm{MPC}} \equiv \cS^\star$. Because $\mathfrak{S}_B$ partitions $\R_{>0}^T$ into finitely many cones defined by homogeneous linear inequalities in $\tilde{w}$, altering $\cS^\star$ requires changing ratios $\tilde{w}_j / \tilde{w}_{j'}$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:shrinkage} (Midpoint Chord-Shrinkage)}
\label{app:proof_thm3}

\begin{proof}
Fix $\cS \in \mathfrak{S}_B$. Under $\cR_\alpha(\cB_{s_m}, s_m + k) = v_{s_m} + \alpha \Delta_{s_m}$, substituting into Eq.~\eqref{eq:adjoint_sum} gives $\delta\Phi(\alpha) = u_0 + \alpha u_1 \in \cH$, where $u_0$ is the $\alpha=0$ perturbation and $u_1 = J_{\Phi \circ \cD}(x_T^\star)\sum_{m=0}^{B-2}\Psi_{s_{m+1}\to T}(\sum_{k=1}^{g_m-1}\omega_{m,k})\Delta_{s_m}$. Expanding the Hilbert norm yields
\begin{equation}
\begin{aligned}
\mathcal{E}(\alpha) &= \E\norm{u_0 + \alpha u_1}_2^2 \\
&= \E\norm{u_0}_2^2 + 2\alpha\E\langle u_0, u_1\rangle + \alpha^2\E\norm{u_1}_2^2 \\
&= \mathcal{E}_{\min} + c_2(\alpha - \alpha^\star)^2,
\end{aligned}
\end{equation}
with $c_2 = \E\norm{u_1}_2^2 > 0$ and $\alpha^\star = -\E\langle u_0, u_1\rangle / c_2$, proving Part~1.

For Part~2, over $\mathcal{I}_m = [s_m, s_{m+1})$ with $a_\ell \approx 1$, $b_{s_m+k} \approx \bar{b}_m$, and $v_{s_m+k}^\star - v_{s_m}^\star = k\dot{v}_{s_m} + \mathcal{O}(k^2 \norm{\ddot{v}_{s_m}}_2)$, using $\sum_{k=1}^{g_m-1} k = \frac{(g_m-1)g_m}{2}$ gives the interval state drift
\begin{equation}
\begin{aligned}
\Delta x_m(\alpha) &= \bar{b}_m \sum_{k=1}^{g_m - 1} (\alpha \Delta_{s_m} - k\dot{v}_{s_m}) + \mathcal{O}(g_m^3 \norm{\ddot{v}_{s_m}}_2) \\
&= \bar{b}_m(g_m - 1)\!\left(\alpha \Delta_{s_m} - \frac{g_m}{2}\dot{v}_{s_m}\right) \\
&\quad + \mathcal{O}(g_m^3 \norm{\ddot{v}_{s_m}}_2).
\end{aligned}
\end{equation}
Minimizing $\E\norm{\alpha \Delta_{s_m} - \frac{g_m}{2}\dot{v}_{s_m}}_2^2$ wrt $\alpha$ yields $\alpha_m^\star = \frac{g_m}{2h_m}\bar{\rho}_m$ with $\bar{\rho}_m = \frac{h_m \E\langle \dot{v}_{s_m}, \Delta_{s_m}\rangle}{\E\norm{\Delta_{s_m}}_2^2}$, proving Eq.~\eqref{eq:midpoint_alpha}.

For Part~3, under $\cR_{\mathrm{scale}}(s_m+k) = v_{s_m} + k\Delta_{s_m}$ ($h_m=1$, $\Delta_{s_m} = \dot{v}_{s_m} - \frac{1}{2}\ddot{v}_{s_m} + \xi_m$), the step-$k$ error is $-\frac{k(k+1)}{2}\ddot{v}_{s_m} + k\xi_m + \mathcal{O}(k^3)$. Summing over $k = 1, \dots, g_m - 1$ gives
\begin{equation}
\begin{aligned}
\Delta x_m(\cR_{\mathrm{scale}}) = \bar{b}_m \Bigl[&-\frac{(g_m - 1)g_m(g_m + 1)}{6}\ddot{v}_{s_m} \\
&+ \frac{(g_m - 1)g_m}{2}\xi_m\Bigr],
\end{aligned}
\end{equation}
up to $\mathcal{O}(g_m^4)$, scaling as $\Theta(g_m^3 \norm{\ddot{v}_{s_m}}_2)$ and inflating step-level variance by $k_{\mathrm{rms}}^2(\cS) = \frac{1}{T-B}\sum_m \sum_{k=1}^{g_m-1} k^2$.
\end{proof}
