\section{Complete Mathematical Proofs}
\label{app:proofs}

In this appendix, we provide complete derivations and proofs for Proposition~\ref{prop:subspace}, Theorem~\ref{thm:kkt}, Corollary~\ref{cor:sloppy}, Theorem~\ref{thm:invariance}, Theorem~\ref{thm:certify}, and Theorem~\ref{thm:shrinkage}.

\subsection{Proof of Proposition~\ref{prop:subspace} (Adjoint Decomposition)}
\label{app:proof_prop1}

\begin{proof}
Let $\{x_i^\star\}_{i=0}^T$ satisfy $x_{i+1}^\star = a_i x_i^\star + b_i v_i^\star$ with $v_i^\star = f_\omega(x_i^\star, \tau_i, c)$, and let $\{x_i\}_{i=0}^T$ be generated under schedule $\cS = \{s_0, \dots, s_{B-1}\}$ ($s_0 = 0, s_{B-1} = T-1$) and predictor $\cR$, satisfying $x_{i+1} = a_i x_i + b_i \tilde{v}_i$ with $x_0 = x_0^\star$. Set $\delta x_i \coloneqq x_i - x_i^\star$ and $J_i \coloneqq \nabla_x f_\omega(x_i^\star, \tau_i, c)$.

\textbf{Step 1 (Closed-loop horizon unrolling).}
Fix an interval $\mathcal{I}_m = \{s_m, \dots, s_{m+1} - 1\}$ of gap $g_m = s_{m+1} - s_m \ge 1$. At exact step $i = s_m \in \cS$, first-order Taylor expansion around $x_{s_m}^\star$ gives
\begin{equation}
\tilde{v}_{s_m} - v_{s_m}^\star = J_{s_m} \delta x_{s_m} + \mathcal{O}(\norm{\delta x_{s_m}}_2^2),
\end{equation}
so $\delta x_{s_m + 1} = (a_{s_m} I + b_{s_m} J_{s_m})\delta x_{s_m} + \mathcal{O}(\norm{\delta x_{s_m}}_2^2) = A_{s_m}^{\mathrm{cl}} \delta x_{s_m} + \mathcal{O}(\norm{\delta x_{s_m}}_2^2)$. Across skipped steps $j = s_m + k$ ($1 \le k \le g_m - 1$), $\cR(\cB_{s_m}, j)$ depends only on cached anchors up to $s_m$, so unrolling $x_{j+1} - x_{j+1}^\star = a_j \delta x_j + b_j(\cR(\cB_{s_m}, j) - v_j^\star)$ across $k = 1, \dots, g_m - 1$ yields
\begin{equation}
\begin{aligned}
\delta x_{s_{m+1}} &= \Bigl(\prod_{\ell = s_m + 1}^{s_{m+1}-1} a_\ell\Bigr) A_{s_m}^{\mathrm{cl}} \delta x_{s_m} \\
&\quad + \sum_{k=1}^{g_m - 1} \omega_{m,k} \bigl(\cR(\cB_{s_m}, s_m + k) - v_{s_m + k}^\star\bigr),
\end{aligned}
\end{equation}
up to $\mathcal{O}(\norm{\delta x}_2^2)$, with $\omega_{m,k} = (\prod_{\ell=s_m+k+1}^{s_{m+1}-1} a_\ell) b_{s_m+k}$. Inducting over $m = 0, \dots, B - 2$ with $\delta x_0 = 0$ and closed-loop transition $\Psi_{s_{m+1}\to T} = \prod_{\ell=s_{m+1}}^{T-1} A_\ell^{\mathrm{cl}}$ yields Eq.~\eqref{eq:adjoint_sum}, and the chain rule on $\Phi \circ \cD$ at $x_T^\star$ gives $\Phi(\cD(x_T)) - \Phi(\cD(x_T^\star)) = J_{\Phi \circ \cD}(x_T^\star)\,\delta x_T + \mathcal{O}(\norm{\delta x_T}_2^2)$.

\textbf{Step 2 (Subspace orthogonality bound).}
Let $\mathcal{U}_m \coloneqq \mathrm{span}\{v : (\cdot, v) \in \cB_{s_m}\} \subseteq \R^d$. Decomposing $v_j^\star = \Pi_{\cB_{s_m}} v_j^\star + \Pi_{\cB_{s_m}}^\perp v_j^\star$ orthogonally in $\R^d$, for any $\cR_{\mathrm{lin}}(\cB_{s_m}, j) \in \mathcal{U}_m$ the Pythagorean theorem gives
\begin{equation}
\begin{aligned}
\norm{\cR_{\mathrm{lin}}(\cB_{s_m}, j) - v_j^\star}_2^2 &= \norm{\cR_{\mathrm{lin}}(\cB_{s_m}, j) - \Pi_{\cB_{s_m}} v_j^\star}_2^2 \\
&\quad + \norm{\Pi_{\cB_{s_m}}^\perp v_j^\star}_2^2 \ge \norm{\Pi_{\cB_{s_m}}^\perp v_j^\star}_2^2,
\end{aligned}
\end{equation}
with equality iff $\cR_{\mathrm{lin}}(\cB_{s_m}, j) = \Pi_{\cB_{s_m}} v_j^\star$, proving Eq.~\eqref{eq:subspace_floor}.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:kkt} (KKT Head \& Geometric Tail)}
\label{app:proof_thm1}

\begin{proof}
Fix $\beta > 0$, $B \in (0, T)$, and continuous strictly decreasing $\tilde{w} : [0, T] \to (0, \infty)$. For $f(\rho) \coloneqq \frac{1}{1+\beta}(\rho^{-\beta} - \rho)$ on $(0, 1]$:
\begin{equation}
f'(\rho) = -\frac{\beta \rho^{-(1+\beta)} + 1}{1 + \beta} < 0, \qquad f''(\rho) = \frac{\beta}{\rho^{2+\beta}} > 0.
\end{equation}
Since $\tilde{w}(t) > 0$, $\tilde{w}(t) f(\rho(t))$ is strictly convex, diverges as $\rho \downarrow 0$ (enforcing $\rho^\star(t) > 0$), and vanishes at $\rho = 1$. With dual multiplier $\lambda \in \R$ for $\int_0^T \rho(t)\,dt = B$ and $\mu(t) \ge 0$ for $\rho(t) - 1 \le 0$, necessary and sufficient KKT conditions are:
\begin{align}
-\frac{\tilde{w}(t)}{1+\beta}\bigl(\beta \rho^\star(t)^{-(1+\beta)} + 1\bigr) + \lambda^\star + \mu^\star(t) &= 0, \label{eq:kkt_stat}\\
0 < \rho^\star(t) \le 1,\;\; \mu^\star(t) \ge 0,\;\; \mu^\star(t)\bigl(\rho^\star(t) - 1\bigr) &= 0. \label{eq:kkt_cs}
\end{align}

\textbf{Part 1 (Head saturation and budget monotonicity).}
Where $\rho^\star(t) < 1$, complementary slackness gives $\mu^\star(t) = 0$, so
\begin{equation}
\beta \tilde{w}(t) \rho^\star(t)^{-(1+\beta)} = (1+\beta)\lambda^\star - \tilde{w}(t).
\end{equation}
Having $\rho^\star(t) \in (0, 1)$ requires $(1+\beta)\lambda^\star - \tilde{w}(t) > \beta\tilde{w}(t) \iff \tilde{w}(t) < \lambda^\star$, yielding $\rho^\star(t) = \bigl(\frac{\beta\tilde{w}(t)}{(1+\beta)\lambda^\star - \tilde{w}(t)}\bigr)^{\frac{1}{1+\beta}}$. Where $\tilde{w}(t) \ge \lambda^\star$, setting $\rho^\star(t) = 1$ and $\mu^\star(t) = \tilde{w}(t) - \lambda^\star \ge 0$ satisfies Eqs.~\eqref{eq:kkt_stat}--\eqref{eq:kkt_cs}. Since $\tilde{w}$ is strictly decreasing, $\{t : \tilde{w}(t) \ge \lambda^\star\} = [0, H^\star(B)]$. Moreover, $G(\lambda) \coloneqq \int_0^T \rho^\star(t; \lambda)\,dt$ is continuous and strictly decreasing for $\lambda \in (\tilde{w}(T), \infty)$ with $G(\tilde{w}(0)) \le T$ and $\lim_{\lambda \to \infty} G(\lambda) = 0$. Hence $\lambda^\star(B)$ is unique and strictly decreasing in $B \in (0, T)$, so $H^\star(B) = \min(T, \max(0, \tilde{w}^{-1}(\lambda^\star(B))))$ is non-decreasing in $B$. As $\beta \downarrow 0$, $f(\rho) \to 1 - \rho$ degenerates to $\rho^\star(t) = \mathbb{I}[t \le B]$.

\textbf{Part 2 (Growing tail).}
For $t > H^\star(B)$, $g^\star(t) = 1/\rho^\star(t) = \bigl(\frac{1+\beta}{\beta}\frac{\lambda^\star}{\tilde{w}(t)} - \frac{1}{\beta}\bigr)^{\frac{1}{1+\beta}}$ increases strictly with $t$. When $\tilde{w}(t) = w_0 e^{-\eta t} \ll (1+\beta)\lambda^\star$, $g^\star(t) \simeq C e^{\frac{\eta}{1+\beta} t}$, giving discrete ratio $g_{k+1}/g_k \approx e^{\eta/(1+\beta)} = r > 1$.
\end{proof}

\subsection{Proof of Corollary~\ref{cor:sloppy} (Stiff-Sloppy Anisotropy)}
\label{app:proof_cor1}

\begin{proof}
Let $\rho \in L^\infty([0, T]; (0, 1])$ satisfy $\int_0^T \rho(t)\,dt = B$. By convexity of $f(\rho)$ and KKT stationarity~\eqref{eq:kkt_stat}, Taylor expansion around $\rho^\star$ gives the exact Bregman decomposition
\begin{equation}
\begin{aligned}
\mathcal{J}(\rho) - \mathcal{J}(\rho^\star) &= \int_0^{H^\star(B)} \bigl(\tilde{w}(t) - \lambda^\star\bigr)\bigl(1 - \rho(t)\bigr)\,dt \\
&\quad + \frac{\beta}{2}\int_0^T \frac{\tilde{w}(t)}{\tilde{\rho}(t)^{2+\beta}}\bigl(\rho(t) - \rho^\star(t)\bigr)^2\,dt,
\end{aligned}
\end{equation}
where we used $\tilde{w}(t) f'(\rho^\star(t)) = -\lambda^\star - \mu^\star(t)$, $\int_0^T \lambda^\star(\rho - \rho^\star)\,dt = 0$, and $\mu^\star(t) = 0$ on $(H^\star(B), T]$.

1.~\emph{Stiff head perturbation:} If $\rho_H$ forces skips on $[H, H^\star(B)]$ ($H = H^\star(B) - \Delta$, $\rho_H(t) \le \rho_{\max} < 1$), the active KKT multiplier $\mu^\star(t) = \tilde{w}(t) - \lambda^\star > 0$ yields a strictly positive \emph{first-order} penalty
\begin{equation}
\begin{aligned}
\mathcal{J}(\rho_H) - \mathcal{J}(\rho^\star) &\ge (1 - \rho_{\max})\int_H^{H^\star} \bigl(\tilde{w}(t) - \lambda^\star\bigr)\,dt \\
&= \Theta(\Delta\,\bar{\mu}_{\mathrm{head}}).
\end{aligned}
\end{equation}

2.~\emph{Sloppy tail perturbation:} If $\rho_\delta$ keeps $\rho_\delta(t) = 1$ on $[0, H^\star(B)]$ and perturbs tail decay $\eta \mapsto \eta(1+\delta)$ while preserving $\int_{H^\star}^T \rho_\delta(t)\,dt = B - H^\star(B)$, then $\rho_\delta(t) - \rho^\star(t) = -\frac{\delta \eta}{1+\beta}(t - \bar{t})\rho^\star(t) + \mathcal{O}(\delta^2)$ on $(H^\star, T]$ (with $\rho^\star$-centroid $\bar{t}$). Substituting $\tilde{w}(t) \approx \frac{1+\beta}{\beta}\lambda^\star \rho^\star(t)^{1+\beta}$ yields
\begin{equation}
\begin{aligned}
\mathcal{J}(\rho_\delta) - \mathcal{J}(\rho^\star) &= \frac{\delta^2 \eta^2 \lambda^\star}{2(1+\beta)}\int_{H^\star(B)}^T (t - \bar{t})^2 \rho^\star(t)\,dt \\
&\quad + \mathcal{O}(\delta^3),
\end{aligned}
\end{equation}
which vanishes to first order and is suppressed by small tail density $\rho^\star(t) \ll 1$ and $\lambda^\star \ll \tilde{w}(0)$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:invariance} (Scale-Invariance Collapse)}
\label{app:proof_thm2}

\begin{proof}
We induct on the exact call index $k = 0, \dots, B - 2$. At $k=0$, $s_0 = 0 = s_0^\star$. Assume after $k$ exact calls the online controller matches $(s_0^\star, \dots, s_k^\star)$. At step $s_k^\star$ with $B - k$ calls left, let $\mathfrak{S}_{B-k}(s_k^\star)$ be strictly increasing sequences $(u_0, \dots, u_{B-1-k})$ with $u_0 = s_k^\star, u_{B-1-k} = T-1$. Since $\hat{\sigma}_k(x_{0:s_k^\star}) > 0$ is constant wrt $(u_1, \dots, u_{B-2-k})$,
\begin{equation}
\begin{aligned}
&\arg\min_{u \in \mathfrak{S}_{B-k}(s_k^\star)} \sum_{m=0}^{B-2-k} \hat{\sigma}_k\, C_\beta(u_m, u_{m+1}) \\
&\qquad = \arg\min_{u \in \mathfrak{S}_{B-k}(s_k^\star)} \sum_{m=0}^{B-2-k} C_\beta(u_m, u_{m+1}).
\end{aligned}
\end{equation}
By Bellman's Principle of Optimality on $\cS_B^\star$, $(s_k^\star, \dots, s_{B-1}^\star)$ solves the unscaled suffix problem, so $u_1 = s_{k+1}^\star$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:certify} (Conformal Budget Certification)}
\label{app:proof_thm3_cert}

\begin{proof}
Let $\{c_1, \dots, c_n, c_{\mathrm{new}}\}$ be exchangeable prompts drawn from $\mathcal{P}$, with $\kappa(c) \coloneqq \mathrm{LPIPS}(\cS_{\mathrm{cal}}, 0; c) / \cC_\beta(\cS_{\mathrm{cal}})$. By exchangeability of $\{\kappa(c_1), \dots, \kappa(c_n), \kappa(c_{\mathrm{new}})\}$, the rank of $\kappa(c_{\mathrm{new}})$ among the $n+1$ variables is uniform on $\{1, \dots, n+1\}$, giving
\begin{equation}
\mathbb{P}\bigl[\kappa(c_{\mathrm{new}}) \le \hat{\kappa}_{1-\delta}\bigr] \ge \frac{\lceil (n+1)(1-\delta)\rceil}{n+1} \ge 1 - \delta.
\end{equation}
On $\{\kappa(c_{\mathrm{new}}) \le \hat{\kappa}_{1-\delta}\}$, evaluating at $\cS_{B^\star}^\star$ ($V_{B^\star}(T-1) = \cC_\beta(\cS_{B^\star}^\star)$) yields
\begin{equation}
\begin{aligned}
\mathrm{LPIPS}(\cS_{B^\star}^\star, \alpha; c_{\mathrm{new}}) &\le \kappa(c_{\mathrm{new}})\,\rho(\alpha)\,V_{B^\star}(T-1) + \xi_{\mathrm{res}} \\
&\le \hat{\kappa}_{1-\delta}\,\rho(\alpha)\,V_{B^\star}(T-1) + \xi_{\mathrm{res}} \\
&\le \eps + \xi_{\mathrm{res}},
\end{aligned}
\end{equation}
by Eq.~\eqref{eq:cert_budget}. Since $\tilde{w}_j > 0$, $B \mapsto V_B(T-1)$ is strictly decreasing on $\{2, \dots, T\}$, so $\rho(\alpha) < 1$ implies $B^\star(\eps, \delta, \alpha) \le B^\star(\eps, \delta, 0)$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:shrinkage} (Phase-Gated Midpoint Shrinkage)}
\label{app:proof_thm4}

\begin{proof}
Fix $\cS \in \mathfrak{S}_B$. On Regime~II ($s_m \ge H_{\mathrm{crit}}$), $\cR_{\alpha, H_{\mathrm{crit}}}(\cB_{s_m}, s_m + k) = v_{s_m} + \alpha \Delta_{s_m}$, so Eq.~\eqref{eq:adjoint_sum} gives $\delta\Phi(\alpha) = u_0 + \alpha u_1 \in \cH$. Expanding $\mathcal{E}(\alpha) = \E\norm{u_0 + \alpha u_1}_2^2$ gives $\mathcal{E}_{\min} + c_2(\alpha - \alpha^\star)^2$ with $c_2 = \E\norm{u_1}_2^2 > 0$ and $\alpha^\star = -\E\langle u_0, u_1\rangle / c_2$, proving Part~1.

For Part~2, over $\mathcal{I}_m = [s_m, s_{m+1})$ with $a_\ell \approx 1$, $b_{s_m+k} \approx \bar{b}_m$, and $\sum_{k=1}^{g_m-1} k = \frac{(g_m-1)g_m}{2}$:
\begin{equation}
\begin{aligned}
\Delta x_m(\alpha) &= \bar{b}_m(g_m - 1)\!\left(\alpha \Delta_{s_m} - \frac{g_m}{2}\dot{v}_{s_m}\right) \\
&\quad + \mathcal{O}(g_m^3 \norm{\ddot{v}_{s_m}}_2),
\end{aligned}
\end{equation}
minimized wrt $\alpha$ at $\alpha_m^\star = \frac{g_m}{2h_m}\bar{\rho}_m$ with $\bar{\rho}_m = \frac{h_m \E\langle \dot{v}_{s_m}, \Delta_{s_m}\rangle}{\E\norm{\Delta_{s_m}}_2^2}$.

For Part~3, under gap-scaling $v_{s_m} + k\Delta_{s_m}$ ($h_m=1$), summing $-\frac{k(k+1)}{2}\ddot{v}_{s_m} + k\xi_m$ over $k=1..g_m-1$ yields
\begin{equation}
\begin{aligned}
\Delta x_m(\cR_{\mathrm{scale}}) = \bar{b}_m\Bigl[&-\frac{(g_m - 1)g_m(g_m + 1)}{6}\ddot{v}_{s_m} \\
&+ \frac{(g_m - 1)g_m}{2}\xi_m\Bigr],
\end{aligned}
\end{equation}
scaling as $\Theta(g_m^3 \norm{\ddot{v}_{s_m}}_2)$ with variance multiplier $k_{\mathrm{rms}}^2(\cS) = \frac{1}{T-B}\sum_m\sum_{k=1}^{g_m-1}k^2$. Finally, in Regime~I ($j < H_{\mathrm{crit}}$) where $\E\langle \dot{v}_{s_m}, \Delta_{s_m}\rangle \le 0$, $\E\norm{\alpha\Delta_{s_m} - \frac{g_m}{2}\dot{v}_{s_m}}_2^2$ is strictly increasing on $\alpha > 0$, so gating $g_t = 0$ strictly lowers interval MSE while preserving $\cR_\alpha$ whenever $H^\star(B) \ge H_{\mathrm{crit}}$.
\end{proof}
